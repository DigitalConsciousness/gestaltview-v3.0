-- ── AI Training & Agent Pipeline ─────────────────────────────────────────

CREATE TABLE public.agents (
agent_id uuid NOT NULL DEFAULT gen_random_uuid(),
slug text NOT NULL UNIQUE,
title text NOT NULL,
domain text NOT NULL,
owner_user_id uuid,
status text NOT NULL CHECK (status = ANY (ARRAY['draft'::text, 'reviewed'::text, 'approved'::text, 'deployed'::text, 'archived'::text])),
active_version_id uuid,
created_at timestamp with time zone NOT NULL DEFAULT now(),
updated_at timestamp with time zone NOT NULL DEFAULT now(),
CONSTRAINT agents_pkey PRIMARY KEY (agent_id),
CONSTRAINT agents_owner_user_id_fkey FOREIGN KEY (owner_user_id) REFERENCES auth.users(id),
CONSTRAINT agents_active_version_fk FOREIGN KEY (active_version_id) REFERENCES public.agent_versions(version_id)
);

CREATE TABLE public.agent_versions (
version_id uuid NOT NULL DEFAULT gen_random_uuid(),
agent_id uuid NOT NULL,
parent_version_id uuid,
source_run_id uuid,
semantic_version text NOT NULL,
canonical_spec jsonb NOT NULL,
compiled_markdown text NOT NULL,
checksum text NOT NULL,
change_summary text,
status text NOT NULL CHECK (status = ANY (ARRAY['candidate'::text, 'approved'::text, 'rejected'::text, 'deployed'::text])),
created_at timestamp with time zone NOT NULL DEFAULT now(),
CONSTRAINT agent_versions_pkey PRIMARY KEY (version_id),
CONSTRAINT agent_versions_agent_id_fkey FOREIGN KEY (agent_id) REFERENCES public.agents(agent_id),
CONSTRAINT agent_versions_parent_version_id_fkey FOREIGN KEY (parent_version_id) REFERENCES public.agent_versions(version_id)
);

CREATE TABLE public.training_runs (
run_id uuid NOT NULL DEFAULT gen_random_uuid(),
agent_id uuid NOT NULL,
baseline_version_id uuid,
requested_by uuid,
approver_user_id uuid,

goal text NOT NULL,
max_cycles integer NOT NULL DEFAULT 3,
quality_threshold numeric NOT NULL,
routing_policy jsonb NOT NULL DEFAULT '{}'::jsonb,
started_at timestamp with time zone,
completed_at timestamp with time zone,
created_at timestamp with time zone NOT NULL DEFAULT now(),
CONSTRAINT training_runs_pkey PRIMARY KEY (run_id),
CONSTRAINT training_runs_approver_user_id_fkey FOREIGN KEY (approver_user_id) REFERENCES auth.users(id),
CONSTRAINT training_runs_agent_id_fkey FOREIGN KEY (agent_id) REFERENCES public.agents(agent_id),
CONSTRAINT training_runs_baseline_version_id_fkey FOREIGN KEY (baseline_version_id) REFERENCES public.agent_versions(version_id),
CONSTRAINT training_runs_requested_by_fkey FOREIGN KEY (requested_by) REFERENCES auth.users(id)
);

CREATE TABLE public.training_steps (
step_id uuid NOT NULL DEFAULT gen_random_uuid(),
run_id uuid NOT NULL,
cycle_no integer NOT NULL,
stage text NOT NULL CHECK (stage = ANY (ARRAY['normalize'::text, 'curriculum'::text, 'scenario_expand'::text, 'author'::text, 'evaluate'::text, 'critique'::text, 'safety'::text, 'package'::text])),
provider_id uuid,
model_id uuid,
request_payload jsonb NOT NULL DEFAULT '{}'::jsonb,
response_payload jsonb,
latency_ms integer,
estimated_cost_usd numeric,
status text NOT NULL CHECK (status = ANY (ARRAY['running'::text, 'completed'::text, 'failed'::text, 'skipped'::text])),
error_message text,
created_at timestamp with time zone NOT NULL DEFAULT now(),
CONSTRAINT training_steps_pkey PRIMARY KEY (step_id),
CONSTRAINT training_steps_run_id_fkey FOREIGN KEY (run_id) REFERENCES public.training_runs(run_id),
CONSTRAINT training_steps_provider_id_fkey FOREIGN KEY (provider_id) REFERENCES public.model_providers(provider_id),
CONSTRAINT training_steps_model_id_fkey FOREIGN KEY (model_id) REFERENCES public.models(model_id)
);

CREATE TABLE public.models (
model_id uuid NOT NULL DEFAULT gen_random_uuid(),
provider_id uuid NOT NULL,
slug text NOT NULL UNIQUE,
api_name text NOT NULL,
modality text NOT NULL DEFAULT 'text'::text,
supports_structured boolean NOT NULL DEFAULT false,
supports_tools boolean NOT NULL DEFAULT false,
supports_embeddings boolean NOT NULL DEFAULT false,
context_window integer,
speed_tier smallint NOT NULL DEFAULT 2,
cost_tier smallint NOT NULL DEFAULT 1,
enabled boolean NOT NULL DEFAULT true,
metadata jsonb NOT NULL DEFAULT '{}'::jsonb,
created_at timestamp with time zone NOT NULL DEFAULT now(),
CONSTRAINT models_pkey PRIMARY KEY (model_id),
CONSTRAINT models_provider_id_fkey FOREIGN KEY (provider_id) REFERENCES public.model_providers(provider_id)
);

-- ── Users & Auth ──────────────────────────────────────────────────────────

CREATE TABLE public.users (
id uuid NOT NULL,
email text NOT NULL UNIQUE,
tier text NOT NULL DEFAULT 'free'::text CHECK (tier = ANY (ARRAY['free'::text, 'core'::text, 'pro'::text, 'enterprise'::text])),
stripe_customer_id text UNIQUE,
stripe_subscription_id text,
subscription_status text DEFAULT 'inactive'::text CHECK (subscription_status = ANY (ARRAY['active'::text, 'inactive'::text, 'past_due'::text, 'canceled'::text, 'trialing'::text])),
billing_period_start timestamp with time zone,
billy_query_count integer NOT NULL DEFAULT 0,
created_at timestamp with time zone NOT NULL DEFAULT now(),
updated_at timestamp with time zone NOT NULL DEFAULT now(),
is_admin boolean NOT NULL DEFAULT false,
grace_until timestamp with time zone,
CONSTRAINT users_pkey PRIMARY KEY (id),
CONSTRAINT users_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id)
);
